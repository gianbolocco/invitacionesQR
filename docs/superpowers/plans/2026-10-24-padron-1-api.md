# Padrón y seguro del auto — fase 1: API

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Pasar el seguro del proveedor de la invitación a un padrón por DNI, sumar el seguro del auto por patente y congelar en `entry_log` el estado de los dos seguros en cada ingreso.

**Architecture:** Dos tablas nuevas, `provider` (clave: DNI normalizado) y `vehicle` (clave: patente normalizada), sin FKs desde la invitación. La garita cruza por texto normalizado. `canEnter` sigue siendo puro y recibe el estado de los dos seguros ya resuelto. El servicio nuevo `services/padron.ts` concentra las lecturas y escrituras del padrón. Lo usan el check, el ingreso, las correcciones y el link público.

**Tech Stack:** Express 5 + zod, Drizzle ORM (Postgres 16), drizzle-kit para migraciones, vitest + supertest.

**Spec:** `docs/superpowers/specs/2026-10-24-padron-seguros-contactos-design.md`

## Global Constraints

- Código, nombres y comentarios en español (salvo el esquema de la base y las rutas). Comentarios que explican el porqué.
- Hora de Buenos Aires siempre: `todayInBuenosAires()` de `api/src/lib/dates.ts`. Nunca `new Date()` para calcular un día.

- Tests: `beforeEach(async () => { await resetDb(); resetRateLimits() })`.

```ts
import { sql, type AnyColumn, type SQL } from 'drizzle-orm'

/*
 * El padrón cruza por texto, no por FK: el DNI y la patente se guardan como
 * los escribió cada uno ("30.123.456", "ab 123 cd") y se comparan normalizados.
 * Todo lo que entra al padrón y toda búsqueda en él pasa por acá.
 */

/** Solo dígitos y letras ASCII, en mayúscula: '30.123.456' → '30123456'. */
export const normalizarDoc = (s: string) => s.toUpperCase().replace(/[^0-9A-Z]/g, '')

/** La misma regla: 'ab-123 cd' → 'AB123CD'. */
export const normalizarPatente = normalizarDoc

/** normalizarDoc en SQL, para cruzar una columna cruda (invitation.guest_doc) contra el padrón. */
export const docSql = (col: AnyColumn | SQL) =>
  sql`upper(regexp_replace(coalesce(${col}, ''), '[^0-9A-Za-z]', '', 'g'))`
```

```ts
export type ClaveSeguro = 'no_aplica' | 'sin_cargar' | 'vencido' | 'por_vencer' | 'vigente'

/**
 * El estado de un seguro, para el proveedor y para el auto. Es lo que se
 * congela en entry_log al ingresar: el padrón cambia, la auditoría no.
 */
export function claveSeguro(pide: boolean, s: Seguro | null, warnDays: number | null, now: Date = new Date()): ClaveSeguro {
  if (!pide) return 'no_aplica'
  if (!s) return 'sin_cargar'
  if (!seguroVigente(s.expiresOn, now)) return 'vencido'
  const dias = (Date.parse(s.expiresOn) - Date.parse(todayInBuenosAires(now))) / 86_400_000
  return warnDays && dias <= warnDays ? 'por_vencer' : 'vigente'
}

const ETIQUETA_SEGURO: Record<ClaveSeguro, 'No aplica' | 'Sin cargar' | 'Vencido' | 'Por vencer' | 'Vigente'> = {
  no_aplica: 'No aplica', sin_cargar: 'Sin cargar', vencido: 'Vencido', por_vencer: 'Por vencer', vigente: 'Vigente',
}

/** Lo que se lee en una planilla. Mismas reglas que estadoSeguro del front. */
export function estadoSeguro(
  r: { requiresInsurance: boolean; company: string | null; policy: string | null; expiresOn: string | null },
  warnDays: number | null,
  now: Date = new Date(),
) {
  const s = seguroCompleto(r.company, r.policy, r.expiresOn)
    ? { company: r.company!, policy: r.policy!, expiresOn: r.expiresOn! } : null
  return ETIQUETA_SEGURO[claveSeguro(r.requiresInsurance, s, warnDays, now)]
}

/** Etiqueta de una clave congelada en entry_log; null (filas viejas o carga manual) es '—'. */
export const etiquetaClave = (c: string | null) => (c ? ETIQUETA_SEGURO[c as ClaveSeguro] : '—')
```

```ts
export const providers = pgTable('provider', {
```

```ts
import { and, desc, eq, isNotNull } from 'drizzle-orm'
import { db } from '../db/index.js'
import { entryLogs, people, providers, units, vehicles } from '../db/schema.js'
import { audit } from '../lib/audit.js'
import { AppError } from '../lib/errors.js'
import { docSql, normalizarDoc, normalizarPatente } from '../lib/padron.js'
import type { Seguro } from '../lib/seguro.js'

/** db o una transacción: con tx, lo que se escribe se revierte junto con el ingreso. */
export type Executor = typeof db | Parameters<Parameters<typeof db.transaction>[0]>[0]

export type ProveedorPadron = {
  id: string
  name: string
  insurance: Seguro | null
  insuranceSource: 'garita' | 'link' | null
  updatedAt: Date | null
}

export type VehiculoPadron = {
  id: string
  plate: string
  insurance: Seguro
  /** El guardia que lo cargó: es lo que el veredicto muestra ("lo cargó Julio el 02/10"). */
  updatedByName: string | null
  updatedAt: Date
}

const seguroDe = (r: { insuranceCompany: string | null; insurancePolicy: string | null; insuranceExpiresOn: string | null }): Seguro | null =>
  r.insuranceCompany && r.insurancePolicy && r.insuranceExpiresOn
    ? { company: r.insuranceCompany, policy: r.insurancePolicy, expiresOn: r.insuranceExpiresOn }
    : null

export async function buscarProveedor(neighborhoodId: string, doc: string, tx: Executor = db): Promise<ProveedorPadron | null> {
  const clave = normalizarDoc(doc)
  if (!clave) return null
  const [r] = await tx.select().from(providers)
    .where(and(eq(providers.neighborhoodId, neighborhoodId), eq(providers.doc, clave))).limit(1)
  if (!r) return null
  return {
    id: r.id, name: r.name, insurance: seguroDe(r),
    insuranceSource: r.insuranceSource as ProveedorPadron['insuranceSource'], updatedAt: r.insuranceUpdatedAt,
  }
}

export async function buscarVehiculo(neighborhoodId: string, plate: string, tx: Executor = db): Promise<VehiculoPadron | null> {
  const clave = normalizarPatente(plate)
  if (!clave) return null
  const [r] = await tx.select({ v: vehicles, updatedByName: people.name }).from(vehicles)
    .leftJoin(people, eq(people.id, vehicles.insuranceUpdatedBy))
    .where(and(eq(vehicles.neighborhoodId, neighborhoodId), eq(vehicles.plate, clave))).limit(1)
  if (!r) return null
  return { id: r.v.id, plate: r.v.plate, insurance: seguroDe(r.v)!, updatedByName: r.updatedByName, updatedAt: r.v.insuranceUpdatedAt }
}

/**
 * La última patente con la que entró este DNI. Sale de entry_log y no de una
 * tabla persona↔auto: es una sugerencia para el guardia, no un vínculo.
 *
 * ponytail: regexp sobre guest_doc sin índice, scan de los ingresos del barrio.
 * Con decenas de miles de filas duele: índice de expresión sobre docSql(guest_doc).
 */
export async function ultimaPatente(neighborhoodId: string, doc: string): Promise<string | null> {
  const clave = normalizarDoc(doc)
  if (!clave) return null
  const [r] = await db.select({ plate: entryLogs.plate }).from(entryLogs)
    .innerJoin(units, eq(units.id, entryLogs.unitId))
    .where(and(eq(units.neighborhoodId, neighborhoodId), eq(docSql(entryLogs.guestDoc), clave), isNotNull(entryLogs.plate)))
    .orderBy(desc(entryLogs.enteredAt)).limit(1)
  return r?.plate ?? null
}

const columnasSeguro = (s: Seguro) => ({
  insuranceCompany: s.company.trim(),
  insurancePolicy: s.policy.trim(),
  insuranceExpiresOn: s.expiresOn,
})

/**
 * Alta o actualización de un proveedor. Sin `insurance` toca solo el nombre y
 * el rubro: un ingreso sin seguro nuevo no borra el que había.
 */
export async function guardarProveedor(tx: Executor, p: {
  neighborhoodId: string
  doc: string
  name: string
  tradeId?: string | null
  insurance?: Seguro
  source?: 'garita' | 'link'
  actorId: string | null
}): Promise<void> {
  const doc = normalizarDoc(p.doc)
  if (!doc) throw new AppError(400, 'doc_required')
  const antes = p.insurance ? await buscarProveedor(p.neighborhoodId, doc, tx) : null

  const seguro = p.insurance ? {
    ...columnasSeguro(p.insurance),
    insuranceSource: p.source ?? 'garita',
    insuranceUpdatedBy: p.actorId,
    insuranceUpdatedAt: new Date(),
  } : {}
  const rubro = p.tradeId !== undefined ? { tradeId: p.tradeId } : {}

  const [fila] = await tx.insert(providers)
    .values({ neighborhoodId: p.neighborhoodId, doc, name: p.name.trim(), ...rubro, ...seguro })
    .onConflictDoUpdate({ target: [providers.neighborhoodId, providers.doc], set: { name: p.name.trim(), ...rubro, ...seguro } })
    .returning({ id: providers.id })

  if (p.insurance) {
    await audit(p.actorId, p.neighborhoodId, 'provider.insurance_updated', 'provider', fila.id,
      { doc, antes: antes?.insurance ?? null, despues: p.insurance }, tx)
  }
}

/** Alta o actualización del seguro de un auto. Lo carga siempre la garita, con el certificado a la vista. */
export async function guardarVehiculo(tx: Executor, p: {
  neighborhoodId: string
  plate: string
  insurance: Seguro
  actorId: string | null
}): Promise<void> {
  const plate = normalizarPatente(p.plate)
  if (!plate) throw new AppError(400, 'patente_requerida')
  const antes = await buscarVehiculo(p.neighborhoodId, plate, tx)

  const valores = { ...columnasSeguro(p.insurance), insuranceUpdatedBy: p.actorId, insuranceUpdatedAt: new Date() }
  const [fila] = await tx.insert(vehicles)
    .values({ neighborhoodId: p.neighborhoodId, plate, ...valores })
    .onConflictDoUpdate({ target: [vehicles.neighborhoodId, vehicles.plate], set: valores })
    .returning({ id: vehicles.id })

  await audit(p.actorId, p.neighborhoodId, 'vehicle.insurance_updated', 'vehicle', fila.id,
    { plate, antes: antes?.insurance ?? null, despues: p.insurance }, tx)
}
```

```ts
 * de la invitación. El veredicto se recalcula acá, con el único canEnter, cada
 * vez que el guardia cambia el documento o la patente.
 */
async function buildCheck(invitation: LoadedInvitation, datos: { doc?: string; plate?: string } = {}) {
  const { usedCount, lastEntryAt } = await usage(invitation.id)
  const { providerWindows, insuranceWarnDays, neighborhoodId, vehicleInsuranceRequired, vehicleInsuranceOverride, ...crudo } = invitation
  const doc = datos.doc ?? invitation.guestDoc ?? ''
  const plate = datos.plate ?? invitation.plate ?? ''
  const [provider, vehicle, lastPlate] = await Promise.all([
    buscarProveedor(neighborhoodId, doc),
    buscarVehiculo(neighborhoodId, plate),
    ultimaPatente(neighborhoodId, doc),

  const check: EntryCheck = canEnter(inv, new Date(), usedCount, providerWindows)
  const abierta = await openEntry(invitation.id)

  return {
    invitation: inv,

    franjas: franjasDelRubro(providerWindows, invitation.kind, invitation.frequentAs, invitation.tradeId),
    insuranceWarnDays,
    check,
    usedCount,
    lastEntryAt,
    provider: provider && { insurance: provider.insurance, insuranceSource: provider.insuranceSource, updatedAt: provider.updatedAt },
    vehicle: vehicle && { plate: vehicle.plate, insurance: vehicle.insurance, updatedByName: vehicle.updatedByName, updatedAt: vehicle.updatedAt },
    lastPlate,
    vehicleInsuranceRequired,
    vehicleInsuranceOverride,
    /*
     * Si viene, el próximo movimiento es un egreso y `check` no aplica: un
     * egreso no se valida. El cliente decide a qué endpoint pegar con esto, y
     * el servidor valida igual por si el dato quedó viejo.

  }
}
```

```ts
export async function registerEntry(
  invitationId: string,
  guardId: string | null,
  data: { guestName: string; guestDoc?: string; plate?: string; note?: string; insurance?: Seguro; vehicleInsurance?: Seguro },
) {
  return db.transaction(async (tx) => {
    const [target] = await tx.select().from(invitations)
      .where(eq(invitations.id, invitationId)).for('update').limit(1)
    if (!target) throw new AppError(404, 'not_found')

    const { usedCount } = await usage(invitationId, tx)
    // Después del FOR UPDATE: quién la creó decide si canEnter mira franjas y seguro.
    const [{ creatorRole }] = await tx.select({ creatorRole: people.role }).from(people).where(eq(people.id, target.createdBy))

    // Después del FOR UPDATE de arriba, que sigue siendo lo que serializa el cupo.
    const [barrio] = await tx.select({
      providerWindows: neighborhoods.providerWindows, nid: neighborhoods.id, warnDays: neighborhoods.insuranceWarnDays,
      vehicleInsuranceRequired: neighborhoods.vehicleInsuranceRequired, vehicleInsuranceOverride: neighborhoods.vehicleInsuranceOverride,
    }).from(units).innerJoin(neighborhoods, eq(neighborhoods.id, units.neighborhoodId))
      .where(eq(units.id, target.unitId)).limit(1)
    const [rubro] = target.tradeId
      ? await tx.select({ requiresInsurance: trades.requiresInsurance }).from(trades).where(eq(trades.id, target.tradeId))
      : []
    const pide = pideSeguro(target.kind, target.frequentAs, rubro?.requiresInsurance ?? null)
    const doc = data.guestDoc?.trim() ?? ''
    const plate = data.plate?.trim() ?? ''

    // El padrón se escribe ANTES de canEnter y en esta misma transacción: si
    // canEnter rechaza, el throw lo revierte junto con su auditoría.
    if (aplicaFranjas(target.kind, target.frequentAs) && doc) {
      await guardarProveedor(tx, {
        neighborhoodId: barrio.nid, doc, name: data.guestName, tradeId: target.tradeId,
        insurance: pide ? data.insurance : undefined, actorId: guardId,
      })
    }
    if (data.vehicleInsurance) {
      await guardarVehiculo(tx, { neighborhoodId: barrio.nid, plate, insurance: data.vehicleInsurance, actorId: guardId })
    }

    const provider = await buscarProveedor(barrio.nid, doc, tx)
    const vehicle = await buscarVehiculo(barrio.nid, plate, tx)
    const inv = aLike({ ...target, creatorRole, requiresInsurance: rubro?.requiresInsurance ?? null }, {
      provider, vehicle, hayPatente: normalizarPatente(plate) !== '',
      vehicleInsuranceRequired: barrio.vehicleInsuranceRequired, vehicleInsuranceOverride: barrio.vehicleInsuranceOverride,
    })
    const ahora = new Date()
    const check = canEnter(inv, ahora, usedCount, barrio.providerWindows)
    if (!check.ok) throw new AppError(409, check.reason)

    const [entry] = await tx.insert(entryLogs).values({
      invitationId,
      unitId: target.unitId,
      guardId,
      guestName: data.guestName.trim(),
      guestDoc: doc || null,
      plate: plate.toUpperCase() || null,
      note: data.note?.trim() || null,
      // La foto de los seguros en este momento: el padrón cambia, la auditoría no.
      // Se calcula sin las excepciones de la portería: así se ve lo que se autorizó igual.
      providerInsurance: claveSeguro(pide, provider?.insurance ?? null, barrio.warnDays, ahora),
      vehicleInsurance: claveSeguro(inv.vehicleRequiresInsurance, vehicle?.insurance ?? null, barrio.warnDays, ahora),
    }).returning()

    // Si la invitación no tenía documento, el que se tipeó en la puerta
    // queda ahí: la próxima vez (frecuente, proveedor) ya viene precargado.
    // La fila de la invitación ya está bloqueada por el FOR UPDATE de arriba.
    if (!target.guestDoc && doc) {
      await tx.update(invitations).set({ guestDoc: doc }).where(eq(invitations.id, invitationId))
    }

    return entry
  })
}
```

```ts
export async function fillGuestDetails(
  token: string,
  data: { guestDoc?: string; plate?: string; insurance?: Seguro },
): Promise<void> {
  const [inv] = await db.select({ inv: invitations, nid: units.neighborhoodId }).from(invitations)
    .innerJoin(units, eq(units.id, invitations.unitId))
    .where(eq(invitations.token, token)).limit(1)
    .then((r) => r.map(({ inv, nid }) => ({ ...inv, nid })))
  if (!inv) throw new AppError(404, 'not_found')
  if (inv.revokedAt) throw new AppError(409, 'revoked')

  // Una vez que entró, entry_log ya guardó su copia: documento y patente no se
  // reescriben. El seguro sí se puede renovar: es lo que el proveedor frecuente necesita.
  if ((data.guestDoc !== undefined || data.plate !== undefined) && (await entriesOf(inv.id)) > 0) {
    throw new AppError(409, 'already_entered')
  }
  // El link lo tiene cualquiera: no puede cargar una póliza que ya venció.
  if (data.insurance && data.insurance.expiresOn < todayInBuenosAires()) throw new AppError(400, 'seguro_vencido')
  await assertSeguro(inv.kind, inv.frequentAs, inv.tradeId, data.insurance)

  const patch: { guestDoc?: string; plate?: string } = {}
  if (!inv.guestDoc && data.guestDoc?.trim()) patch.guestDoc = data.guestDoc.trim()
  if (!inv.plate && data.plate?.trim()) patch.plate = data.plate.trim().toUpperCase()

  // El seguro vive en el padrón, por DNI: sin DNI no hay a quién asignarlo.
  // Vale el de la invitación; si no tenía, el que manda el proveedor ahora.
  const doc = inv.guestDoc ?? patch.guestDoc
  if (data.insurance && !doc) throw new AppError(400, 'doc_required')

  await db.transaction(async (tx) => {
    if (Object.keys(patch).length) await tx.update(invitations).set(patch).where(eq(invitations.id, inv.id))
    if (data.insurance) {
      await guardarProveedor(tx, {
        neighborhoodId: inv.nid, doc: doc!, name: inv.guestName, tradeId: inv.tradeId,
        insurance: data.insurance, source: 'link', actorId: null,
      })
    }
  })
}
```
